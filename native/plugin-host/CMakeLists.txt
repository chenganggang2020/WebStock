cmake_minimum_required(VERSION 3.22)
project(WebStockPluginHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test Network Sql)
enable_testing()
add_library(webstock_plugins STATIC ../desktop/quant_plugin_host.cpp ../desktop/quant_plugin_widget.cpp ../desktop/quant_plugin_form.cpp ../desktop/quant_plugin_center.cpp)
target_sources(webstock_plugins PRIVATE ../desktop/market_history.cpp ../desktop/market_store.cpp ../desktop/market_data.cpp ../desktop/market_feed.cpp ../desktop/tencent_detail.cpp ../desktop/tencent_detail_request.cpp)
target_include_directories(webstock_plugins PUBLIC ../desktop)
target_link_libraries(webstock_plugins PUBLIC Qt6::Widgets Qt6::Network Qt6::Sql)
add_executable(WebStockPluginHost main.cpp)
target_link_libraries(WebStockPluginHost PRIVATE webstock_plugins)
add_executable(quant_plugin_tests ../tests/quant_plugin_tests.cpp)
target_link_libraries(quant_plugin_tests PRIVATE webstock_plugins Qt6::Test)
add_test(NAME quant_plugin_platform COMMAND quant_plugin_tests)
