cmake_minimum_required(VERSION 3.22)
project(WebStockPluginHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test)
enable_testing()
add_library(webstock_plugins STATIC ../desktop/quant_plugin_host.cpp ../desktop/quant_plugin_widget.cpp)
target_include_directories(webstock_plugins PUBLIC ../desktop)
target_link_libraries(webstock_plugins PUBLIC Qt6::Widgets)
add_executable(WebStockPluginHost main.cpp)
target_link_libraries(WebStockPluginHost PRIVATE webstock_plugins)
add_executable(quant_plugin_tests ../tests/quant_plugin_tests.cpp)
target_link_libraries(quant_plugin_tests PRIVATE webstock_plugins Qt6::Test)
add_test(NAME quant_plugin_platform COMMAND quant_plugin_tests)
