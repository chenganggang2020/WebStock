cmake_minimum_required(VERSION 3.22)
project(MarketNativeReview VERSION 0.61.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(domain_tests tests/domain_tests.cpp)
target_include_directories(domain_tests PRIVATE domain-cpp)
add_test(NAME domain COMMAND domain_tests "${CMAKE_CURRENT_SOURCE_DIR}/contracts")
add_executable(legacy_indicators_tests tests/legacy_indicators_tests.cpp)
target_include_directories(legacy_indicators_tests PRIVATE domain-cpp)
add_test(NAME legacy_indicators COMMAND legacy_indicators_tests "${CMAKE_CURRENT_SOURCE_DIR}/contracts")
option(BUILD_REVIEW_UI "Build native Qt review UI" OFF)
if(BUILD_REVIEW_UI)
  add_subdirectory(desktop)
endif()
